{{ TP4 iteracion: repeat con pout y una invocacion dentro del cuerpo.
   El cuerpo tiene varias sentencias, por lo que va entre begin y end
   (un bloque_sentencias puede ser una sola sentencia o un begin-end).
   label_1 es el inicio del ciclo. La invocacion genera PAR y CALL,
   y el resultado se imprime con pout.
   Esperado:
   1. (:=, x, 0$ul)
   2. (label_1, --, --)
   3. (PAR, n, x)
   4. (CALL, foo, 1)
   5. (:=, r, [4])
   6. (pout, r, --)
   7. (+, x, 1$ul)
   8. (:=, x, [7])
   9. (<, x, 3$ul)
   10. (BF, [9], label_2)
   11. (BI, label_1, --)
   12. (label_2, --, --)
   Sin errores. }}
prog
ulongint function foo(ulongint n)
begin
    ret(n);
end;
ulongint x, r;
begin
    x := 0$ul;
    repeat
    begin
        r := foo(x)[1$ul];
        pout(r);
        x := x + 1$ul;
    end
    while (x < 3$ul);
end
