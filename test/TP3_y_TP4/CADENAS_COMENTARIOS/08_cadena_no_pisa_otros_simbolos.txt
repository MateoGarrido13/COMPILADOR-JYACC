{{ TP4 cadenas y comentarios: la cadena no comparte clave con otros
   simbolos. Una cadena puede tener el mismo texto que una palabra
   reservada o que una constante numerica; cada entrada debe conservar
   su token, tipo y uso.
   Esperado:
   [303] Cadena {if}
   [303] Cadena {1$ul}
   [302] Constante ulongint 1$ul
   if -> 400    Tipo de dato: indefinido   Uso: reservada
   if -> 303    Tipo de dato: cadena       Uso: cadena
   1$ul -> 303  Tipo de dato: cadena       Uso: cadena
   1$ul -> 302  Tipo de dato: ulongint     Uso: constante
   2. (pout, if, --)
   3. (pout, 1$ul, --)
   4. (:=, x:main, 1$ul)
   Sin errores. }}
prog
ulongint x;
begin
    pout({if});
    pout({1$ul});
    x := 1$ul;
end
