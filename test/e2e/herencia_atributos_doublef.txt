class a
begin
    ulongint x_a;
end;
class b
begin
    ulongint x_b;
    extends a;
end;
class c
begin
    ulongint x_c;
    extends b;
end;
c obj;
ulongint u;
doublef d;
begin
    obj.c.x_c := 3$ul;
    obj.b.x_b := 2$ul;
    u := obj.c.x_c;
    pout(u);
    u := obj.b.x_b;
