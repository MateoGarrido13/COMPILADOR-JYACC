   Las etiquetas internas se reservan antes. label_1 abre el else

   9. (label_2, --, --)

ulongint a, b, k;
begin
    if (a > 0$ul)
        if (b > 0$ul)
            k := 1$ul;
        else
            k := 2$ul;
        end_if;
    else
        k := 3$ul;
    end_if;
