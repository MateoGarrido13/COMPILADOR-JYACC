   ca y cb tienen el mismo unico atributo ulongint, pero distinto tipo de
   retorno en m. La asignacion es valida y genera (COPY, a1:main, b1:main).
   Sin errores.
}}
prog
class ca
begin
    ulongint x;
    ulongint m(ulongint n)
    begin
        ret(n);
    end;
end;
class cb
begin
    ulongint x;
    doublef m(ulongint n)
    begin
        ret(1.0);
    end;
end;
ca a1;
cb b1;

    a1 := b1;
