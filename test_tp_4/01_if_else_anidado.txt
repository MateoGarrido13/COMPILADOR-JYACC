prog
ulongint x, y, z;
begin
    x := 1$ul;
    y := 2$ul;
    z := 0$ul;
    if (x < y)
        if (z == 0$ul)
            z := x + y;
        else
            z := y - x;
        end_if;
    else
        z := 9$ul;
    end_if;
    pout(z);
end
