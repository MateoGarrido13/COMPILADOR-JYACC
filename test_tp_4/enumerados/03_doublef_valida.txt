{{ TP4 5.h: asignacion VALIDA a enum de base doublef.
   El valor viene de una variable (no de un literal), por lo que el control
   estatico de Verificaciones no lo rechaza: debe chequearse en runtime con
   FLD / FCOMP / FSTSW / SAHF / JZ contra c_1_5 y c_2_5.
   FCOMP hace POP, asi que la pila FPU queda balanceada. }}
prog
typedef medidas = [1.5, 2.5];
medidas m;
doublef v;
begin
    v := 1.5;
    m := v;
    pout(m);
end
