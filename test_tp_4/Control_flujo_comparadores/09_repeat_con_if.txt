{{ TP4a: repeat con if dentro del cuerpo.
    Verifica: labels de repeat (inicio/salida) y labels de if conviven,
    BI de if salta a etiqueta propia, BI de repeat salta a inicio.
}}
prog
ulongint i, s;
begin
    i := 1$ul;
    s := 0$ul;
    repeat
    begin
        if (i < 5$ul)
        begin
            s := s + i;
            pout(s);
        end
        end_if;
        i := i + 1$ul;
        pout(i);
    end
    while (i <= 10$ul);
end
